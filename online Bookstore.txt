/* =========================================
   ONLINE BOOKSTORE DATABASE PROJECT
   ========================================= */

/* =========================================
   1. CREATE TABLES
   ========================================= */

CREATE TABLE Authors (
    author_id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    bio TEXT
);

CREATE TABLE Genres (
    genre_id INT PRIMARY KEY,
    name VARCHAR(50) NOT NULL
);

CREATE TABLE Books (
    book_id INT PRIMARY KEY,
    title VARCHAR(200) NOT NULL,
    author_id INT,
    price DECIMAL(10,2),
    publication_date DATE,
    genre_id INT,
    FOREIGN KEY (author_id) REFERENCES Authors(author_id),
    FOREIGN KEY (genre_id) REFERENCES Genres(genre_id)
);

CREATE TABLE Customers (
    customer_id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100),
    phone VARCHAR(20)
);

CREATE TABLE Orders (
    order_id INT PRIMARY KEY,
    customer_id INT,
    order_date DATE,
    total DECIMAL(10,2),
    FOREIGN KEY (customer_id)
        REFERENCES Customers(customer_id)
);

CREATE TABLE Order_Details (
    order_id INT,
    book_id INT,
    quantity INT,
    PRIMARY KEY (order_id, book_id),
    FOREIGN KEY (order_id)
        REFERENCES Orders(order_id),
    FOREIGN KEY (book_id)
        REFERENCES Books(book_id)
);

/* =========================================
   3. CREATE VIEWS
   ========================================= */

/* Books by Author */
CREATE VIEW Books_By_Author AS
SELECT
    a.author_id,
    a.name AS author_name,
    b.book_id,
    b.title
FROM Authors a
JOIN Books b
ON a.author_id = b.author_id;

/* Total Sales Per Book */
CREATE VIEW Book_Sales AS
SELECT
    b.book_id,
    b.title,
    SUM(od.quantity) AS total_quantity_sold,
    SUM(od.quantity * b.price) AS total_sales_amount
FROM Books b
JOIN Order_Details od
ON b.book_id = od.book_id
GROUP BY b.book_id, b.title;

/* Customer Order History */
CREATE VIEW Customer_Order_History AS
SELECT
    c.customer_id,
    c.name,
    o.order_id,
    o.order_date,
    o.total
FROM Customers c
JOIN Orders o
ON c.customer_id = o.customer_id;

/* =========================================
   4. QUERIES
   ========================================= */

/* Retrieve all books with authors and genres */

SELECT
    b.book_id,
    b.title,
    a.name AS author_name,
    g.name AS genre_name,
    b.price,
    b.publication_date
FROM Books b
JOIN Authors a
ON b.author_id = a.author_id
JOIN Genres g
ON b.genre_id = g.genre_id;

/* Top 5 Best-Selling Books */

SELECT
    b.book_id,
    b.title,
    SUM(od.quantity) AS total_sold
FROM Books b
JOIN Order_Details od
ON b.book_id = od.book_id
GROUP BY b.book_id, b.title
ORDER BY total_sold DESC
LIMIT 5;

/* Orders with Details and Customer Information */

SELECT
    o.order_id,
    c.name AS customer_name,
    c.email,
    o.order_date,
    b.title,
    od.quantity,
    b.price,
    (od.quantity * b.price) AS subtotal
FROM Orders o
JOIN Customers c
ON o.customer_id = c.customer_id
JOIN Order_Details od
ON o.order_id = od.order_id
JOIN Books b
ON od.book_id = b.book_id;

/* =========================================
   5. INDEXES
   ========================================= */

CREATE INDEX idx_books_author
ON Books(author_id);

CREATE INDEX idx_books_genre
ON Books(genre_id);

CREATE INDEX idx_orders_customer
ON Orders(customer_id);

CREATE INDEX idx_orderdetails_order
ON Order_Details(order_id);

CREATE INDEX idx_orderdetails_book
ON Order_Details(book_id);

/* =========================================
   6. OPTIMIZATION EXAMPLES
   ========================================= */

/* Less Efficient */

SELECT *
FROM Books
WHERE author_id = 1;

/* Better with Index */

SELECT
    book_id,
    title,
    price
FROM Books
WHERE author_id = 1;

/* Find Customers Who Never Ordered */

SELECT
    c.customer_id,
    c.name
FROM Customers c
LEFT JOIN Orders o
ON c.customer_id = o.customer_id
WHERE o.order_id IS NULL;

/* Highest Selling Book */

SELECT
    b.title,
    SUM(od.quantity) AS total_sales
FROM Books b
JOIN Order_Details od
ON b.book_id = od.book_id
GROUP BY b.book_id, b.title
ORDER BY total_sales DESC
LIMIT 1;
